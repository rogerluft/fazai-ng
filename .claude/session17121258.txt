       const [loading, setLoading] = useState(false);

       const fetchZones = async () => {
         setLoading(true);
         const response = await fetch('/api/cloudflare/zones');
         const data = await response.json();
         setZones(data.data);
         setLoading(false);
       };

       return { zones, loading, fetchZones };
     }
