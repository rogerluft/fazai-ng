import os


def gerar_lista_inodes(diretorio):

    inodes = []

    for arquivo in os.listdir(diretorio):

        caminho = os.path.join(diretorio, arquivo)

        if os.path.isfile(caminho):

            inode = os.stat(caminho).st_ino

            inodes.append((arquivo, inode))

    return inodes
